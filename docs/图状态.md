
# SciTraceState

```python
from typing import Literal

from langgraph.graph import MessagesState
from pydantic import BaseModel

from scitrace.models.experiment import ExperimentSpec
from scitrace.models.execution import ExperimentRun
from scitrace.models.resource import ResearchResource


class RouteDirective(BaseModel):
    """SciTraceAgent 当前尚未履行的确定性路由约束。"""

    target: Literal[
        "discovery",
        "analysis",
    ]
    reason: str


class SciTraceState(MessagesState):
    """SciTraceAgent 在一次 Task 执行过程中的主工作状态。"""

    # 业务工作状态
    task_id: str
    resources: list[ResearchResource]
    experiment_spec: ExperimentSpec | None = None
    experiment_run: ExperimentRun | None = None

    # Agent Loop 控制状态，满足一定条件的就要强制路由到对应的agent
    route: RouteDirective | None = None

    execution_attempts: int = 0
```

# DiscoveryState

```python
class DiscoveryAgentState(MessagesState):
    """DiscoveryAgent 单次调用期间的内部状态。"""

    task_id: str
    resources: list[ResearchResource]
    recovery_count: int
```

# AnalysisState

```python
class AnalysisAgentState(MessagesState):
    """AnalysisAgent 单次调用期间的内部工作状态。"""

    task_id: str

    resources: list[ResearchResource]

    experiment_spec: ExperimentSpec | None = None

    experiment_run: ExperimentRun | None = None

    recovery_count: int = 0
```
