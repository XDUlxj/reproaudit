# 1. Task类
```Plain text
Task
│
├── query
│   └── 用户说了什么
│
├── attachments
│   └── 用户额外给了什么
│
└── lifecycle
    ├── status
    └── final_answer
```

```python
class Task(Record):
    """用户提交给 SciTrace 的一次完整科研请求。"""

    query: str # 用户的原始请求
    attachment_ids: list[str] = Field(default_factory=list) # 附件

    status: Literal[
        "created",
        "running",
        "finished",
        "failed",
    ] = "created" # 任务状态

    answer: str | None = None
```

# 2. ResearchResource类

```Plain text
Task
“用户要什么”
        │
        ▼
SciTraceAgent
        │
        ├──── 用户已经提供资源 ────┐
        │                         │
        └──── DiscoveryAgent ─────┤
                                  ▼
                         Resource Tools
                         识别 / 解析 / 标准化
                                  │
                                  ▼
                        ResearchResource
                                  │
             ┌────────┬───────────┼────────┐
             ▼        ▼           ▼        ▼
           Paper     Repo       Dataset   Model
```

```python
from typing import Annotated, Any, Literal

from pydantic import BaseModel, Field

from scitrace.models.base import Record


class WebLocation(BaseModel):
    kind: Literal["web"] = "web"
    url: str


class LocalLocation(BaseModel):
    kind: Literal["local"] = "local"
    path: str


class AttachmentLocation(BaseModel):
    kind: Literal["attachment"] = "attachment"
    attachment_id: str
    filename: str


ResourceLocation = Annotated[
    WebLocation | LocalLocation | AttachmentLocation,
    Field(discriminator="kind"),
]


class ResourceBase(Record):
    """SciTrace 已识别并标准化的科研资源。"""

    kind: Literal[
        "paper",
        "repository",
        "dataset",
        "model",
    ]

    name: str

    locations: list[ResourceLocation] = Field(default_factory=list)

    metadata: dict[str, Any] = Field(default_factory=dict)


class PaperResource(ResourceBase):
    kind: Literal["paper"] = "paper"

    doi: str | None = None
    arxiv_id: str | None = None


class RepositoryResource(ResourceBase):
    kind: Literal["repository"] = "repository"

    revision: str | None = None


class DatasetResource(ResourceBase):
    kind: Literal["dataset"] = "dataset"

    version: str | None = None


class ModelResource(ResourceBase):
    kind: Literal["model"] = "model"

    revision: str | None = None


ResearchResource = Annotated[
    PaperResource
    | RepositoryResource
    | DatasetResource
    | ModelResource,
    Field(discriminator="kind"),
]
```

# 3. ExperimentSpec类

结构：
```Plain text
AnalysisAgent
     │
     │ 已经完成：
     │ - 理解 Task
     │ - 理解论文
     │ - 检查 Repo
     │ - Paper ↔ Code Mapping
     │ - 参数冲突消解
     │ - 定位脚本/config
     │ - 构造最终执行方式
     │
     ▼
ExperimentSpec
     │
     ├── goal
     │   └── 为什么执行
     │
     ├── resource_ids
     │   └── 执行需要什么资源
     │
     ├── commands
     │   └── 具体执行什么
     │
     └── outputs
         └── 要收集什么
     │
     ▼
ExecutionAgent
```

```python
from typing import Annotated, Literal

from pydantic import BaseModel, Field

from scitrace.models.base import Record


class MetricCriterion(BaseModel):
    """标量指标的复现判据。"""

    kind: Literal["metric"] = "metric"

    name: str

    reference_value: float

    operator: Literal[
        "approximately_equal",
        "greater_or_equal",
        "less_or_equal",
    ]

    tolerance: float = Field(default=0.0, ge=0.0)


class ArtifactCriterion(BaseModel):
    """Artifact 的复现判据。"""

    kind: Literal["artifact"] = "artifact"

    name: str

    verifier: Literal[
        "existence",
        "structured",
        "semantic",
    ]

    requirement: str


VerificationCriterion = Annotated[
    MetricCriterion | ArtifactCriterion,
    Field(discriminator="kind"),
]


class ExperimentSpec(Record):
    """AnalysisAgent 交付的正式实验方案。"""

    # 要复现什么
    goal: str

    # 执行实验需要的科研资源
    resource_ids: list[str] = Field(default_factory=list)

    # 实际需要执行的命令
    commands: list[str] = Field(min_length=1)

    # 实验需要产生什么结果，以及这些结果如何验证
    verification_criteria: list[VerificationCriterion] = Field(
        default_factory=list
    )

    # 当前 Spec 如果由另一个 Spec 修订而来，则记录其父 Spec
    parent_spec_id: str | None = None
```

# 4. ExperimentRun类

```python
from typing import Annotated, Literal

from pydantic import BaseModel, Field

from scitrace.models.base import Record


class MetricOutput(BaseModel):
    """实验实际产生的标量指标。"""

    kind: Literal["metric"] = "metric"
    name: str
    value: float


class ArtifactOutput(BaseModel):
    """实验运行过程中产生并持久保存的文件。"""

    kind: Literal["artifact"] = "artifact"
    name: str
    uri: str
    sha256: str


ObservedOutput = Annotated[
    MetricOutput | ArtifactOutput,
    Field(discriminator="kind"),
]

class CommandRun(BaseModel):
    """一条命令的实际执行结果。"""

    command: str

    status: Literal[
        "running",
        "succeeded",
        "failed",
        "interrupted",
    ]

    exit_code: int | None = None
    
class LocalExecutionHandle(BaseModel):
    """本地正在运行的实验进程标识。"""

    pid: int
    create_time: float


class ExperimentRun(Record):
    """一次 ExperimentSpec 的实际执行尝试。"""

    experiment_spec_id: str

    status: Literal[
        "running",
        "succeeded",
        "failed",
        "interrupted",
    ]

    execution_handle: LocalExecutionHandle | None = None

    command_runs: list[CommandRun] = Field(default_factory=list)

    outputs: list[ObservedOutput] = Field(default_factory=list)

    logs: list[ArtifactOutput] = Field(default_factory=list)

    error: str | None = None

    finished_at: str | None = None
```
